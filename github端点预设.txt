# ===== git / GitHub 需要测的端点 =====
# 用法：把本文件内容整段贴到 %APPDATA%\QuickerPing\hosts.txt（或只复制需要的行），
#       然后在动作窗口点「重新检测」。
#
# 语法：
#   host                测 443 的 HTTPS（TLS 握手 + GET，拿到状态码就算通）
#   host:8443           指定端口，同样走 HTTPS
#   tcp/host:22         只测端口通不通（SSH 这类非 HTTPS 端口必须加 tcp/ 前缀）
#   host | 用途 | 通了的说明 | 不通的说明        自定义"这个链接是干嘛的 / 后果"
#   # 开头整行注释；proxy= / icmp= / direct_check= 等是设置项
#
# 下面这些 GitHub 端点，用途和通/不通的后果程序里已经内置，直接列域名就有中文说明。

proxy=auto
icmp=off
direct_check=on      # 走代理时额外测一遍"直连"，用来判断"没开代理时到底能不能开"

# ---- git 走 HTTPS（clone / fetch / pull / push）必需 ----
github.com
api.github.com
codeload.github.com
raw.githubusercontent.com
objects.githubusercontent.com
github.githubassets.com

# ---- git 走 SSH 时（和 HTTPS 二选一即可）----
tcp/github.com:22
tcp/ssh.github.com:443

# ---- 可选：缺了不影响 clone/push ----
# alive.github.com
# collector.github.com
# media.githubusercontent.com
# github-cloud.s3.amazonaws.com
# avatars.githubusercontent.com

# ---- 自己加的域名（示例）----
# x.com | 社交网站 | 可以刷 X | X 打不开
